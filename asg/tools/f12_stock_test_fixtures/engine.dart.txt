Future<void> checkIsolation(Map<String, String> env) async {
  if (PlatformDispatcher.instance.views.isEmpty) {
    throw StateError('No real Flutter engine view');
  }
  final sentinel = File(env['SENTINEL']!);
  try {
    sentinel.readAsStringSync();
    throw StateError('Outside read succeeded');
  } on FileSystemException {
    // Expected denial.
  }
  try {
    sentinel.writeAsStringSync('changed');
    throw StateError('Outside write succeeded');
  } on FileSystemException {
    // Expected denial.
  }
  try {
    await Process.run('/usr/bin/true', []);
    throw StateError('Subprocess started');
  } on ProcessException {
    // Expected denial.
  }
  for (final host in ['127.0.0.1', '::1']) {
    try {
      final canary = await Socket.connect(
        host,
        int.parse(env['CANARY_PORT']!),
        timeout: const Duration(milliseconds: 300),
      );
      canary.destroy();
      throw StateError('Outside connection succeeded');
    } on SocketException {
      // The supervisor also independently checks both canary listeners.
    }
  }
}

const invalidWireCases = {
  'oversized-header',
  'fragmented-header',
  'session-bytes',
  'session-frames',
};

List<int> textHeader(int length) => length < 126
    ? [0x81, 0x80 | length]
    : [0x81, 0xfe, length >> 8, length & 0xff];

Uint8List maskedTextFrame(List<int> header, List<int> body) {
  final frame = Uint8List(header.length + 4 + body.length);
  frame.setRange(0, header.length, header);
  // Fixed mask for these authored test bytes.
  const mask = [0x5a, 0x6b, 0x7c, 0x8d];
  frame.setRange(header.length, header.length + 4, mask);
  for (var i = 0; i < body.length; i++) {
    frame[header.length + 4 + i] = body[i] ^ mask[i % mask.length];
  }
  return frame;
}

Future<void> sendInvalidFrame(Uri uri, String testCase) async {
  final socket = await Socket.connect(uri.host, uri.port);
  final upgraded = Completer<void>();
  final response = <int>[];
  var finalHeaderSent = false;
  final subscription = socket.listen(
    (chunk) {
      if (upgraded.isCompleted) return;
      response.addAll(chunk);
      final text = latin1.decode(response);
      if (text.contains('\r\n\r\n')) {
        if (text.startsWith('HTTP/1.1 101 ')) {
          upgraded.complete();
        } else {
          upgraded.completeError(StateError('Fixture upgrade was refused'));
        }
      }
    },
    onDone: () {
      if (finalHeaderSent) {
        final receipt = '${Platform.environment['HOME']}/wire-peer-closed';
        File('$receipt.tmp')
          ..writeAsStringSync(testCase)
          ..renameSync(receipt);
      }
    },
  );
  try {
    socket.write(
      'GET ${uri.path} HTTP/1.1\r\n'
      'Host: ${uri.authority}\r\n'
      'Connection: Upgrade\r\n'
      'Upgrade: websocket\r\n'
      'Sec-WebSocket-Key: dGhlIHNhbXBsZSBub25jZQ==\r\n'
      'Sec-WebSocket-Version: 13\r\n\r\n',
    );
    await upgraded.future.timeout(const Duration(seconds: 2));
    final List<int> header;
    if (testCase == 'session-bytes' || testCase == 'session-frames') {
      // Valid envelopes on a pending channel keep the suite in its load phase.
      const emptyEnvelope = '[901,{"pad":""}]';
      final large = testCase == 'session-bytes';
      final padding = large ? 'x' * (16384 - emptyEnvelope.length) : '';
      final body = utf8.encode('[901,{"pad":"$padding"}]');
      header = textHeader(body.length);
      final frame = maskedTextFrame(header, body);
      final admitted = large ? 7 : 128;
      for (var i = 0; i < admitted; i++) {
        socket.add(frame);
        await socket.flush();
      }
    } else {
      header = testCase == 'oversized-header'
          ? textHeader(32768)
          : [0x01, 0x80];
    }
    for (var i = 0; i < header.length; i++) {
      finalHeaderSent = i == header.length - 1;
      socket.add([header[i]]);
      await socket.flush();
      await Future<void>.delayed(const Duration(milliseconds: 20));
    }
    // No mask key or payload follows the last declared header.
    await Future<void>.delayed(const Duration(seconds: 10));
  } finally {
    socket.destroy();
    await subscription.cancel();
  }
}

Future<void> main() async {
  final env = Platform.environment;
  final testCase = env['CASE'];
  final host = env['FAMILY'] == 'IPv4' ? '127.0.0.1' : '::1';
  final token = testCase == 'wrong-token' ? 'wrong-token' : env['SESSION'];
  final uri = Uri(
    scheme: 'ws',
    host: host,
    port: int.parse(env['SERVER_PORT']!),
    path: '/$token',
  );
  if (invalidWireCases.contains(testCase)) {
    await sendInvalidFrame(uri, testCase!);
    return;
  }
  final WebSocket socket;
  try {
    socket = await WebSocket.connect(
      uri.toString(),
      compression: CompressionOptions.compressionOff,
    );
  } on WebSocketException {
    if (testCase != 'wrong-token') rethrow;
    // Let the parent observe the denied request and own process termination.
    await Future<void>.delayed(const Duration(seconds: 10));
    return;
  }
  if (testCase == 'missing-suite') {
    await socket.close();
    // Leave process termination to the stock device after the channel closes.
    await Future<void>.delayed(const Duration(seconds: 10));
    return;
  }
  final suite = RemoteListener.start(
    () => () {
      test('fixture renders in confinement', () => checkIsolation(env));
      if (testCase != 'wrong-suite') {
        test('fixture deliberate failure', () {
          if (testCase != 'wrong-result') {
            throw TestFailure('fixture deliberate failure marker');
          }
        });
      }
      if (testCase == 'deadline' || testCase == 'stop') {
        test('fixture waits', () {
          final waiting = '${env['HOME']}/waiting';
          File('$waiting.tmp')
            ..writeAsStringSync(env['FAMILY']!)
            ..renameSync(waiting);
          return Completer<void>().future;
        });
      }
    },
  );
  unawaited(
    socket.map((message) => jsonDecode(message as String)).pipe(suite.sink),
  );
  await socket.addStream(suite.stream.map(jsonEncode));
  // The stock device owns engine termination even if the suite closes first.
  await Future<void>.delayed(const Duration(seconds: 10));
}
